## Usage

```tsx
function BasicExample() {
  const [value, setValue] = useState(1000);

  return (
    <div>
      <RollingNumber value={value} />
      <Button
        size="sm"
        variant="ghost"
        className="border border-primary/10 rounded-full "
        onClick={() => setValue(value + 1000)}
      >
        <Plus className="h-4 w-4" />
      </Button>
    </div>
  );
}
```
